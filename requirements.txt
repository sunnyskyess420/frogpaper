# FrogPaper runtime + build dependencies.
#
# Versions are pinned to specific releases for reproducible builds. Each pin
# is followed by a brief note explaining why that version was chosen. Bump
# intentionally and re-run the test suite before shipping.
#
# Python: FrogPaper targets Python 3.11+. Pillow 13 dropped Python 3.10
# support, and several security advisories patched in Pillow 13.0.0 are
# reachable from this app's image paths (gallery thumbnails, provider
# downloads, cloud-restored images, style transfer, tray icon). See
# SECURITY_NOTES.md for the advisory list and the upgrade rationale.

# --- Core runtime ---

# Security pin. Pillow 13.0.0 patches the 2026-10 advisory wave that affects
# FrogPaper, including:
#   * CVE-2026-96639  multi-frame loaders reuse stale native image storage
#                    after mode transitions (GHSA-v9pv-5gc8-w2m6)
#   * CVE-2026-112403 JPEG decoder heap buffer overflow (ImagingJpegDecode
#                    scaled output / shrinks)
#   * CVE-2026-112406 infinite loop in PdfParser.rewrite_pages
#   * GHSA-78g4-xqwv-q3g8 bundled libopenjp2 2.5.4 heap-buffer-overflow write
#                    reachable from Image.open().load()
#   * GHSA-fc9g-5qg7-mf45 quadratic GIF comment copying during save
#   * GHSA-654x-cwwv-5j9c Arrow export leaks stale heap memory
#   * GHSA-65qm-jvmh-6pqg GIF seek DoS after DecompressionBombError
#   * GHSA-vx6w-vv9f-6gvg EPS load(scale) bypass of decompression bomb
#   * GHSA-xgh8-jp6q-p5x8 ReDoS in XbmImagePlugin
#   * GHSA-jg86-v3jv-hrpm quadratic-time DoS in PDF save_all
#   * GHSA-74r2-7ff9-mjj2 quadratic GIF comment concatenation DoS
#   * GHSA-27h8-gf7j-v5hc NULL deref in ImagingExpand on zero-size images
#   * GHSA-f2g6-542v-9j58 infinite loop in PdfParser.rewrite_pages
# Pillow 13 requires Python 3.11+.
pillow==13.0.0

# Security pin. NLTK 3.10.4 (released 2026-10-21) is the first NLTK release
# with no known security issues per the Snyk advisory database. It fixes:
#   * GHSA-8mgp-746c-j5xp model-artifact APIs bypass pathsec and touch files
#                    outside allowed roots (CVE-2026-81726)
#   * CVE-2026-79676  corpus readers follow symlinks outside trusted roots
# The app only uses the WordNet corpus API; NLTK is excluded from the
# packaged Windows EXE (see FrogPaper.spec excludes=). This pin matters for
# source/dev installs and CI.
nltk==3.10.4

# --- Image / ML ---
# opencv-python MUST be a NumPy 2 build. 4.8.1.78 is compiled against
# NumPy 1.x: importing it under NumPy 2.5.4 fails with
# "numpy.core.multiarray failed to import", which silently killed every
# style filter in the 1.7.6 build. 4.14.0.94 is the NumPy 2 compatible
# release; verify with python -c "import cv2; print(cv2.__version__)"
# after installing, and re-test style_transfer on Windows before changing.
opencv-python==4.14.0.94

# sentence-transformers pulls torch (~2-3 GB). It is an optional, dev-only
# dependency; CI uses a lean install and the packaged EXE excludes it via
# FrogPaper.spec. Pin the upper bound to prevent 7.x breakage.
sentence-transformers>=5.6.0,<7.0.0
